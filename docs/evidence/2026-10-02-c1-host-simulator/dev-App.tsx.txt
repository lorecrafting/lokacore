// c1-host luck replay harness (temporary, never committed; docs/evidence/2026-10-02-c1-host-simulator).
// Each launch opens the dusk save through openStory with the app's real host (expo-crypto random,
// the stamped kernel version) and sends one pick_lock with a fixed invocation id, recording the
// reply's decision bytes and the head rng before and after in the table c1luck.
import { Text, View } from 'react-native';
import { getRandomValues, randomUUID } from 'expo-crypto';
import { openDatabaseSync } from 'expo-sqlite';
import { KERNEL_ID } from '../authority/local-story/session';
import { openStory } from '../authority/local-story/authority';
import { encode } from '../../kernel/ts/src/canonical';
import { INSTALLED, loadCartridge, newWorld } from '../../kernel/ts/src/index';
import dusk from '../../protocol/fixtures/cartridge_dusk_hash.json';

const commit = process.env.EXPO_PUBLIC_KERNEL_COMMIT ?? `${'0'.repeat(40)}-dirty`;
const kernel_version = `${KERNEL_ID}@${commit}${__DEV__ && !commit.endsWith('-dirty') ? '-dirty' : ''}`;

function run(): string {
  const db = openDatabaseSync('c1-dusk.db');
  const artifact = `{"cartridge":${dusk.canonical},"content_hash":"${dusk.sha256}"}`;
  const loaded = loadCartridge(new TextEncoder().encode(artifact), INSTALLED) as any;
  const fresh = newWorld(loaded.cartridge, '0d4e8a5c-3f1b-4c2a-9e7d-6b5a4c3d2e1f' as never, [1, 2, 3, 4] as never);
  const host = { kernel_version, newId: randomUUID, random: getRandomValues as never };
  const story = openStory(db as never, [{ content_hash: dusk.sha256, fresh }], host);
  if (story.kind !== 'open') return `c1luck refused ${story.kind}`;
  const rng = () => (db.getFirstSync('SELECT rng FROM head') as { rng: string }).rng;
  const before = rng();
  const r = story.invoke({
    invocation_id: '00000000-0000-4000-8000-0000000000c1',
    action_key: 'pick_lock',
    actor_id: story.world().character,
    target_ids: [],
    input: {},
  }) as { kind: string; replay?: boolean; decision?: never };
  const after = rng();
  db.execSync('CREATE TABLE IF NOT EXISTS c1luck (n INTEGER PRIMARY KEY, kind, replay, bytes, rng_before, rng_after)');
  const bytes = r.decision ? encode(r.decision) : null;
  db.runSync('INSERT INTO c1luck (kind, replay, bytes, rng_before, rng_after) VALUES (?, ?, ?, ?, ?)', r.kind, String(r.replay), bytes, before, after);
  return `c1luck ${r.kind} replay=${r.replay} before=${before} after=${after}`;
}

const line = run();
export default function App() {
  return (
    <View style={{ flex: 1, justifyContent: 'center', padding: 24 }}>
      <Text>{line}</Text>
    </View>
  );
}
