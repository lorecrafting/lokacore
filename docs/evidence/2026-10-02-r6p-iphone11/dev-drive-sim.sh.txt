#!/bin/zsh
# The Simulator dry run of drive-phone.sh: same phases, same completion and redaction rules.
# Usage: DEV=<simulator> drive-sim.sh <build> <out> <phase...>
source ${0:a:h}/common.sh
setopt pipefail # main's failure (a stop, a timeout) is the script's exit status, through redact
BUILD=$1; OUT=$2; shift 2; APP=com.lorecrafting.loka
export DEVELOPER_DIR=/Applications/Xcode.app/Contents/Developer
function main {
  mkdir -p $OUT
  local data=$(xcrun simctl get_app_container $DEV $APP data)
  local run=$data/Documents/p6run-$BUILD.txt
  for phase in "$@"; do
    local n0=$(lines $run) pid="" got=""
    for t in 1 2 3; do pid=$(xcrun simctl launch $DEV $APP 2>/dev/null | awk '{print $2}'); [[ -n $pid ]] && break; sleep 2; done
    [[ -z $pid ]] && { echo "$phase launch failed; stop"; return 1; }
    for i in {1..120}; do got=$(done_line $run $n0 $phase) && [[ -n $got ]] && break; sleep 1; done
    echo "$phase pid=$pid after line $n0 -> ${got:-TIMEOUT}"
    [[ -z $got ]] && { kill -9 $pid; echo "timeout; stop"; return 1; }
    case $got in *HOLD*)
      mkdir -p $OUT/$phase
      cp $data/Documents/SQLite/p6-$phase.db(N) $data/Documents/SQLite/p6-$phase.db-journal(N) $OUT/$phase/
      ls -la $OUT/$phase | awk 'NR>3 {print $5, $9}' ;;
    esac
    kill -9 $pid; sleep 1
  done
  redact < $run > $OUT/run.txt
}
main "$@" 2>&1 | redact
