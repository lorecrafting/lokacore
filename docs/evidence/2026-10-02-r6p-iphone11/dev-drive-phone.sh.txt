#!/bin/zsh
# Runs harness phases on the phone over the cable: launch, poll the run file (bounded, 120 s) for
# the phase's done/HOLD/ERROR line, copy the held save files BEFORE the kill, kill, next.
# Usage: drive-phone.sh <device> <build> <out> <phase...>
DEV=$1; BUILD=$2; OUT=$3; shift 3; APP=com.lorecrafting.loka
export DEVELOPER_DIR=/Applications/Xcode.app/Contents/Developer
function redact { sed "s#$HOME#<home>#g; s#$DEV#<device>#g"; }
function pull { xcrun devicectl device copy from --device $DEV --domain-type appDataContainer --domain-identifier $APP --source $1 --destination $2 -q 2>&1 | redact; }
mkdir -p $OUT
for phase in "$@"; do
  xcrun devicectl device info lockState --device $DEV 2>&1 | grep -E "passcodeRequired" | redact
  for t in 1 2 3; do
    pid=$(xcrun devicectl device process launch --device $DEV --terminate-existing $APP --json-output - 2>/dev/null | python3 -c 'import json,sys; print(json.load(sys.stdin)["result"]["process"]["processIdentifier"])' 2>/dev/null)
    [[ -n $pid ]] && break; sleep 3
  done
  [[ -z $pid ]] && { echo "$phase launch failed; stop"; exit 1; }
  got=""
  for i in {1..40}; do
    pull Documents/p6run-$BUILD.txt $OUT/run.txt > /dev/null
    if grep -q -E " $phase (done|HOLD|ERROR)" $OUT/run.txt 2>/dev/null; then got=$(grep -E " $phase (done|HOLD|ERROR)" $OUT/run.txt | tail -1); break; fi
    sleep 1
  done
  echo "$phase pid=$pid -> ${got:-TIMEOUT}"; [[ -z $got ]] && { echo "timeout; stop (app left running)"; exit 1; }
  case $got in *HOLD*)
    mkdir -p $OUT/$phase
    for f in p6-$phase.db p6-$phase.db-journal; do pull Documents/SQLite/$f $OUT/$phase/$f; done
    ls -la $OUT/$phase | awk 'NR>1 {print $5, $9}' ;;
  esac
  xcrun devicectl device process terminate --device $DEV --pid $pid --kill -q 2>&1 | redact
  sleep 2
done
