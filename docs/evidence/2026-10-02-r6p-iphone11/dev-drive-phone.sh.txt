#!/bin/zsh
# Runs harness phases on the phone over the cable: note the run file's length, launch, poll it
# (bounded, about 120 s) for this launch's done/HOLD/ERROR line, copy the held save BEFORE the
# kill, kill, next. Usage: DEV=<device> drive-phone.sh <build> <out> <phase...>
source ${0:a:h}/common.sh
BUILD=$1; OUT=$2; shift 2; APP=com.lorecrafting.loka
export DEVELOPER_DIR=/Applications/Xcode.app/Contents/Developer
function pull { rm -f $2; xcrun devicectl device copy from --device $DEV --domain-type appDataContainer --domain-identifier $APP --source $1 --destination $2 -q 2>&1; }
function main {
  mkdir -p $OUT
  local run=$OUT/run.raw
  for phase in "$@"; do
    xcrun devicectl device info lockState --device $DEV 2>&1 | grep passcodeRequired
    pull Documents/p6run-$BUILD.txt $run > /dev/null
    local n0=$(lines $run) pid="" got=""
    for t in 1 2 3; do
      pid=$(xcrun devicectl device process launch --device $DEV --terminate-existing $APP --json-output - 2>/dev/null | python3 -c 'import json,sys; print(json.load(sys.stdin)["result"]["process"]["processIdentifier"])' 2>/dev/null)
      [[ -n $pid ]] && break; sleep 3
    done
    [[ -z $pid ]] && { echo "$phase launch failed; stop"; return 1; }
    for i in {1..40}; do
      pull Documents/p6run-$BUILD.txt $run > /dev/null
      got=$(done_line $run $n0 $phase) && [[ -n $got ]] && break
      sleep 1
    done
    echo "$phase pid=$pid after line $n0 -> ${got:-TIMEOUT}"
    [[ -z $got ]] && { echo "timeout; stop (app left running)"; return 1; }
    case $got in *HOLD*)
      mkdir -p $OUT/$phase
      for f in p6-$phase.db p6-$phase.db-journal; do pull Documents/SQLite/$f $OUT/$phase/$f; done
      ls -la $OUT/$phase | awk 'NR>3 {print $5, $9}' ;;
    esac
    xcrun devicectl device process terminate --device $DEV --pid $pid --kill -q 2>&1
    sleep 2
  done
  redact < $run > $OUT/run.txt # the retained copy
}
main "$@" 2>&1 | redact
