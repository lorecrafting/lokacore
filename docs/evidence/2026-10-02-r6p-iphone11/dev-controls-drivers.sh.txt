#!/bin/zsh
# Red controls for the drivers' stop paths (P6BF-1..3, P6BG-1), with a fake `xcrun` first on PATH that
# answers as a test case says. No device is touched; identifiers are made up.
H=${0:a:h}; T=$(mktemp -d); fail=0
mkdir -p $T/bin $T/tmp $T/simdata/Documents
cat > $T/bin/xcrun <<'X'
#!/bin/zsh
echo "$*" >> $FAKE_DIR/calls
case "$*" in
  *"info lockState"*) echo "• passcodeRequired: false" ;;
  *"copy from"*)
    dest=${@[(i)--destination]}; dest=${@[dest+1]}
    n=$(( $(cat $FAKE_DIR/copies 2>/dev/null || echo 0) + 1 )); echo $n > $FAKE_DIR/copies
    case $FAKE in
      missing_then_stale) # the first copy says missing; later copies hold an old launch's lines
        (( n == 1 )) && { echo "ERROR: Failed to retrieve the file node for Documents/p6run-x.txt on /Application/ABCDEF01-2345-4ABC-8DEF-0123456789AB"; exit 1; }
        print -l "P6RUN p6-1-nold carry2 begin build=x" "P6RUN p6-1-nold carry2 done records=13" "P6RUN p6-1 carry2 done records=13" > $dest ;;
      launch_fail) echo "ERROR: Failed to retrieve the file node for Documents/p6run-x.txt"; exit 1 ;;
      lost_response) cp $FAKE_DIR/simdata/Documents/p6run-x.txt $dest ;;
      good) print -l "P6RUN p6-1-nold carry2 done records=13" "P6RUN p6-2-n$(cat $FAKE_DIR/nonce) carry2 done records=13" > $dest ;;
    esac ;;
  *"process launch"*|*"simctl launch"*)
    l=$(( $(cat $FAKE_DIR/launches 2>/dev/null || echo 0) + 1 )); echo $l > $FAKE_DIR/launches
    if [[ $FAKE == lost_response ]]; then # attempt 1 runs and finishes, but its response is lost
      (( l == 1 )) && { echo "P6RUN p6-1-n${@[-1]} carry2 done records=13" >> $FAKE_DIR/simdata/Documents/p6run-x.txt; exit 0; }
      [[ "$*" == *simctl* ]] && echo "com.lorecrafting.loka: 42" || echo '{"result":{"process":{"processIdentifier":42}}}'; exit 0
    fi
    [[ "$*" == *simctl* ]] && exit 1
    [[ $FAKE == launch_fail ]] && exit 1; echo ${@[-1]} > $FAKE_DIR/nonce; echo '{"result":{"process":{"processIdentifier":42}}}' ;;
  *"simctl get_app_container"*) echo $FAKE_DIR/simdata ;;
esac
X
chmod +x $T/bin/xcrun
function expect { if [[ "$2" == "$3" ]]; then echo "PASS $1"; else echo "FAIL $1: got [$2] want [$3]"; fail=1; fi; }
function run { # run <fake> <driver> -> sets out, code
  rm -f $T/calls $T/copies $T/nonce $T/launches $T/simdata/Documents/p6run-x.txt; rm -rf $T/out $T/tmp/*(N)
  out=$(FAKE=$1 FAKE_DIR=$T TMPDIR=$T/tmp PATH=$T/bin:$PATH DEV=00001111-000A1B2C3D4E5F60 $H/$2 x $T/out carry2 2>&1); code=$?
}
run missing_then_stale drive-phone.sh
expect "phone: missing, then an old launch's done pair: timeout non-zero" "$code $(print -r -- $out | grep -c 'TIMEOUT')" "1 1"
expect "phone: no raw file left after the timeout" "$(find $T/tmp $T/out -name 'run.raw' | wc -l | tr -d ' ')" "0"
expect "phone: the failing run is kept redacted" "$(cat $T/out/run.txt 2>/dev/null | wc -l | tr -d ' ')" "3"
expect "phone: no identifier in the output" "$(print -r -- $out | grep -c ABCDEF01)" "0"
run good drive-phone.sh
expect "phone: this launch's done line: exit 0" "$code $(print -r -- $out | grep -c 'p6-2-n')" "0 1"
run lost_response drive-phone.sh
expect "phone: attempt 1's done line after a lost launch response: retry times out non-zero" "$code $(grep -c 'process launch' $T/calls) $(print -r -- $out | grep -c TIMEOUT)" "1 2 1"
expect "phone: attempt 1 did write its done line (the control is not vacuous)" "$(grep -c ' carry2 done' $T/simdata/Documents/p6run-x.txt)" "1"
run lost_response drive-sim.sh
expect "sim: attempt 1's done line after a lost launch response: retry times out non-zero" "$code $(grep -c 'simctl launch' $T/calls) $(print -r -- $out | grep -c TIMEOUT)" "1 2 1"
expect "sim: attempt 1 did write its done line (the control is not vacuous)" "$(grep -c ' carry2 done' $T/simdata/Documents/p6run-x.txt)" "1"
run launch_fail drive-phone.sh
expect "phone: forced launch failure exits non-zero" "$code $(print -r -- $out | grep -c 'launch failed; stop')" "1 1"
run launch_fail drive-sim.sh
expect "sim: forced launch failure exits non-zero" "$code $(print -r -- $out | grep -c 'launch failed; stop')" "1 1"
rm -rf $T; exit $fail
