# Shared by drive-phone.sh and drive-sim.sh (sourced). Identifiers come from the caller's
# environment (DEV, P6_TEAM), never from this file.
# Every line a driver prints or retains passes through redact: home, scratch paths, the device,
# the team, uppercase UUIDs (CoreDevice ids, simulator and container ids) and the device's name.
function redact {
  sed -E "s#$HOME#<home>#g; s#(/private)?/(tmp|var/folders)/[^ :]*#<scratch>#g; s#${DEV:-<no-device>}#<device>#g; s#${P6_TEAM:-<no-team>}#<team>#g; s#[0-9A-F]{8}-[0-9A-F]{4}-[0-9A-F]{4}-[0-9A-F]{4}-[0-9A-F]{12}#<uuid>#g; s#0000[0-9]{4}-[0-9A-F]{16}#<device>#g; s#([A-Z][^ ]* )*[A-Z][^ ]*[’']s iPhone#<device-name>#g"
}
# done_line <log> <nonce> <phase>: the phase's done/HOLD/ERROR line written by the launch that was
# given <nonce> (its P6RUN id ends in -n<nonce>). A line from any other launch, a retained log or a
# done marker never counts, whatever the copies of the log show.
function done_line {
  grep -E "^P6RUN p6-[0-9]+-n$2 $3 (done|HOLD|ERROR)" $1 2>/dev/null | tail -1
}
function nonce { openssl rand -hex 8; }
