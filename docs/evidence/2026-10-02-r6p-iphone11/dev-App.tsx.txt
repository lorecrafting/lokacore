// R6P P6 device harness entry (evidence only, kept as dev-App.tsx.txt; never committed as App.tsx).
// Each launch runs the next phase of p6/harness.ts on expo-sqlite and appends its lines to
// Documents/p6run-<BUILD>.txt (Release drops console.log). It opens only p6-*.db files.
import { Text, View } from 'react-native';
import { deleteDatabaseSync, openDatabaseSync, type SQLiteDatabase } from 'expo-sqlite';
import { File, Paths } from 'expo/node_modules/expo-file-system';
import { fixtures, PHASES, runPhase, type Env } from './p6/harness';
import { EXPECTED, SIM } from './p6/data';
import items from '../../protocol/fixtures/cartridge_items_hash.json';
import dusk from '../../protocol/fixtures/cartridge_dusk_hash.json';
import lantern from '../../protocol/fixtures/cartridge_lantern_hash.json';
import cases from '../../kernel/ts/test/invocation_cases.json';
import traces from '../../docs/spec/conformance/lantern-traces.json';

const BUILD = 'mutant'; // a red-control build sets its own name, so it starts again at phase 1
const RUN = `p6-${Date.now()}`;
const doc = (name: string) => new File(Paths.document, name);
const sqlite = (name: string) => new File(Paths.document, 'SQLite', name);
const out = doc(`p6run-${BUILD}.txt`);
const emit = (line: string) => {
  if (!out.exists) out.create();
  out.write(`P6RUN ${RUN} ${line}\n`, { append: true });
};
const handles = new Map<string, SQLiteDatabase>();
const env: Env = {
  open(name) {
    if (!name.startsWith('p6-')) throw new Error('not a harness file'); // never the owner's saves
    const db = openDatabaseSync(name);
    handles.set(name, db);
    return db;
  },
  close(name) {
    handles.get(name)?.closeSync();
    handles.delete(name);
  },
  read: (name) => sqlite(name).bytesSync(),
  write(name, bytes) {
    const f = sqlite(name);
    if (!f.exists) f.create();
    f.write(bytes);
  },
  remove(name) {
    if (!name.startsWith('p6-')) throw new Error('not a harness file');
    if (sqlite(name).exists) deleteDatabaseSync(name);
    if (sqlite(`${name}-journal`).exists) sqlite(`${name}-journal`).delete();
  },
  hold(tag) {
    const m = doc('p6-hold.txt');
    if (!m.exists) m.create();
    m.write(`${RUN} ${tag}\n`);
    emit(`${tag} HOLD`);
    const end = Date.now() + 600_000; // bounded: the agent kills within 120 s
    while (Date.now() < end);
    throw new Error('hold not killed');
  },
  emit,
  latency: { host: 'hermes_ios', now: () => performance.now() },
};

function go(): string {
  const done = doc(`p6-done-${BUILD}.txt`);
  const finished = done.exists ? done.textSync().split('\n') : [];
  const phase = PHASES.find((p) => !finished.includes(p));
  if (!phase) return 'all phases done';
  if (!done.exists) done.create();
  done.write(`${phase}\n`, { append: true }); // before running: a held phase is not rerun
  const hp = (globalThis as { HermesInternal?: { getRuntimeProperties?: () => object } })
    .HermesInternal;
  emit(
    `${phase} begin build=${BUILD} hermes=${JSON.stringify(hp?.getRuntimeProperties?.() ?? null)}`,
  );
  fixtures({ items, dusk, lantern, cases, traces } as never);
  try {
    runPhase(env, phase, JSON.parse(SIM), JSON.parse(EXPECTED));
  } catch (e) {
    emit(`${phase} ERROR ${String(e)} ${(e as Error).stack ?? ''}`);
  }
  return `phase ${phase} finished`;
}
const text = go();
export default function App() {
  return (
    <View style={{ flex: 1, justifyContent: 'center', padding: 16 }}>
      <Text>{`${RUN} ${text}`}</Text>
    </View>
  );
}
