import { useEffect, useState } from 'react';
import { openDatabaseAsync, deserializeDatabaseAsync, backupDatabaseAsync, type SQLiteDatabase } from 'expo-sqlite';
import { randomUUID } from 'expo-crypto';
import { openGame } from '../authority/local-story/session.ts';
import { webDb } from './sqlite-web.ts';
import bundle from '../../protocol/fixtures/missing_child_v023_hash.json';
const host = { newId: randomUUID, kernel_version: `loka-kernel@${'0'.repeat(40)}`, time: { wall: () => 10000, monotonic: () => 0 } };
const sqlHistory = `SELECT invocation_id,command_id,actor_id,command,response,revision FROM receipt WHERE json_extract(response,'$.kind')='accepted' ORDER BY revision`;
let opened: SQLiteDatabase;
let game: ReturnType<typeof openGame>;
const digest = async (value: unknown) => Array.from(new Uint8Array(await crypto.subtle.digest('SHA-256', new TextEncoder().encode(JSON.stringify(value)))), (byte) => byte.toString(16).padStart(2, '0')).join('');
async function witness() {
  const db = webDb(opened);
  return { accepted: db.getFirstSync<{ n: number }>("SELECT count(*) AS n FROM receipt WHERE json_extract(response,'$.kind')='accepted'")!.n, head: db.getFirstSync('SELECT revision,clock,rng FROM head'), world: await digest(db.getAllSync('SELECT section,key,value FROM state_row ORDER BY section,key')), projection: await digest(game.view()), narration: game.lastNarration() };
}
export default function Proof() {
  const [report, setReport] = useState('opening');
  const [ready, setReady] = useState(false);
  useEffect(() => { (async () => {
    try {
      opened = await openDatabaseAsync('loka-stream-proof-import.db');
      for (const method of ['getAllSync','getFirstSync','runSync','execSync','isInTransactionSync']) {
        const original = (opened as any)[method].bind(opened);
        (opened as any)[method] = (...args:any[]) => { try { return original(...args); } catch (error) { throw Object.assign(error as Error,{operation:method,sql:args[0]}); } };
      }
      if (!await opened.getFirstAsync("SELECT 1 FROM sqlite_master WHERE name='head'")) { setReport('Empty disposable database: import controlled authority history'); setReady(true); return; }
      const bulk = new URLSearchParams(window.location.search).has('bulk');
      game = openGame(bulk ? opened : webDb(opened), bundle, host);
      opened.execSync('CREATE TABLE IF NOT EXISTS proof_witness (value TEXT)');
      const prior = opened.getFirstSync<{ value: string }>('SELECT value FROM proof_witness');
      const current = await witness();
      if (prior && prior.value !== JSON.stringify(current)) throw new Error('witness changed');
      if (!prior) opened.runSync('INSERT INTO proof_witness VALUES (?)',JSON.stringify(current));
      setReport(JSON.stringify({mode:bulk?'bulk':'stream', coldOpen:'passed', witnessEqual:!!prior, ...current},null,2));
      setReady(!prior);
    } catch (error) { setReport(JSON.stringify({coldOpen:'failed', message:(error as Error).message,operation:(error as any).operation,sql:(error as any).sql,stack:(error as Error).stack})); }
  })(); }, []);
  async function seed() {
    setReady(false);
    try {
      const response = await fetch('/proof.db');
      const source = await deserializeDatabaseAsync(new Uint8Array(await response.arrayBuffer()));
      await backupDatabaseAsync({sourceDatabase:source,destDatabase:opened});
      await source.closeAsync();
      opened.execSync('CREATE TABLE IF NOT EXISTS proof_witness (value TEXT)');
      game = openGame(webDb(opened),bundle,host);
      const bulkBytes = (await (await fetch('/proof-seed.json')).json()).bulkBytes;
      const current = await witness();
      opened.runSync('INSERT INTO proof_witness VALUES (?)',JSON.stringify(current));
      setReport(JSON.stringify({seed:'passed',bulkBytes,...current},null,2));
    } catch(error) { setReport(JSON.stringify({seed:'failed',message:(error as Error).message,operation:(error as any).operation,sql:(error as any).sql,stack:(error as Error).stack})); }
  }
  return <main><h1>SQLite streaming cold-open proof</h1><button disabled={!ready} onClick={seed}>Import 2402 accepted receipts and dialogue</button><p><a href="/?bulk=1">Old bulk cold open</a> · <a href="/">Streaming cold open</a></p><pre>{report}</pre></main>;
}
