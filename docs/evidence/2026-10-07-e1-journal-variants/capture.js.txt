import assert from 'node:assert/strict';
import { mkdirSync, readFileSync, writeFileSync, rmSync } from 'node:fs';
import { join } from 'node:path';
import { source, host, redact } from './kernel/ts/test/e1.ts';
import { admitCandidate, sha256 } from './kernel/ts/test/e1_policy.ts';
import { caseHost } from './kernel/ts/test/e1_case_host.ts';
import { replayCase } from './kernel/ts/test/e1_cases.ts';
import { ending } from './kernel/ts/test/e1_paths.ts';
import { hash } from './kernel/ts/src/foundation/canonical.ts';

// Run from the clean source commit: mise exec -- node --input-type=module - OUTPUT < capture.js.txt
const out = process.argv[2];
const identity = source();
mkdirSync(out);
const pin = JSON.parse(readFileSync('protocol/fixtures/missing_child_v042_hash.json', 'utf8'));
const bytes = new TextEncoder().encode(`{"cartridge":${pin.canonical},"content_hash":"${pin.sha256}"}`);
const loaded = admitCandidate(bytes);
const base = '/quests/ashmere_missing_child@0.0.42:quest/missing_child/journal/active_variants';
const witnesses = new Set();
const cases = [];
for (const child of ['rescued', 'stays']) {
  const id = `${child}-prior`, log = join(out, `${id}.jsonl`), db = join(out, `${id}.db`);
  const a = caseHost(loaded, db, log, `loka-kernel@${identity.source_sha}`, {
    case_id: id, source: identity, fault_schedule: [], certification: false,
  });
  try {
    ending(a, child, 'prior', 'stilled');
    a.record({ kind: 'finish', steps: a.commands.length, digest: a.digest(),
      state_hash: hash(a.story.world().state) });
    const text = readFileSync(log, 'utf8');
    const replay = replayCase(bytes, text, identity);
    const journal = replay.obligations.filter((p) => p.startsWith(base));
    for (const p of journal) witnesses.add(p);
    cases.push({ case_id: id, steps: replay.steps, trace_sha256: sha256(text), journal });
    console.log(JSON.stringify(cases.at(-1)));
  } finally {
    a.close();
    rmSync(db);
  }
}
const expected = [
  '1/when/root', '1/when/root/items/0', '1/when/root/items/1',
  '2/when/root', '2/when/root/items/0', '3/when/root',
  '3/when/root/items/0', '4/when/root', '4/when/root/items/0',
].map((p) => `${base}/${p}`).sort();
assert.deepEqual([...witnesses].sort(), expected);
const journalPaths = loaded.policy.uses.filter((u) => u.feature === 'authored.quest' && u.path.startsWith(base)).map((u) => u.path);
const pending = journalPaths.filter((p) => !witnesses.has(p));
assert.deepEqual(pending, ['0/when/root', '0/when/root/items/0', '0/when/root/items/1'].map((p) => `${base}/${p}`));
writeFileSync(join(out, 'report.json'), `${JSON.stringify({
  status: 'pending', certification_verdict: null, source: identity, host: host(),
  content_hash: loaded.hash, artifact_sha256: sha256(bytes),
  scope: 'two complete prior endings and exact active journal projection witnesses',
  cases, witnessed_journal_paths: [...witnesses].sort(), pending_journal_paths: pending,
  limitations: ['variant 0 separation was not reached', 'two bell objective any children remain outside this slice',
    'no integrated 18-case or 10000-sequence run', 'independent review and E1 final certification remain pending'],
}, null, 2)}\n`);
console.log(redact('Two routes replayed; nine journal paths witnessed; three journal paths pending.'));
