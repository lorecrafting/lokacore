import { Text, View } from 'react-native';
import { File, Paths } from 'expo/node_modules/expo-file-system';
import { INSTALLED, loadCartridge, newWorld } from '../../kernel/ts/src/index.ts';
import { run } from '../../kernel/ts/src/known_answers.ts';
import { cases, dusk_cases } from '../../kernel/ts/test/invocation_cases.json';
import items from '../../protocol/fixtures/cartridge_items_hash.json';
import duskFx from '../../protocol/fixtures/cartridge_dusk_hash.json';

const RUN_ID = `p1-${Date.now()}`;
const world = (kat: any, seed: number[]) => {
  const artifact = `{"cartridge":${kat.canonical},"content_hash":"${kat.sha256}"}`;
  const loaded: any = loadCartridge(new TextEncoder().encode(artifact), INSTALLED);
  if (!loaded.ok) throw new Error('load ' + JSON.stringify(loaded));
  return newWorld(loaded.cartridge, '0d4e8a5c-3f1b-4c2a-9e7d-6b5a4c3d2e1f' as any, seed as any);
};
function go(): string {
  const hp: any = (globalThis as any).HermesInternal;
  const props = hp?.getRuntimeProperties?.() ?? {};
  const out: string[] = [`P1RUN ${RUN_ID} begin`, `P1RUN ${RUN_ID} hermes ${JSON.stringify(props)}`];
  let total = 0;
  const one = (label: string, cs: any[], w: any) => {
    // per-case pass/fail: run each case alone from the same start world
    for (const c of cs) {
      const m = run(w, [c]);
      total += m.length;
      out.push(`P1RUN ${RUN_ID} ${label} ${c.name} ${m.length ? 'FAIL' : 'PASS'}`);
      for (const x of m) out.push(`P1RUN ${RUN_ID} MISMATCH ${x}`);
    }
  };
  try {
    one('items', cases as any[], world(items, [1, 2, 3, 4]));
    one('dusk', dusk_cases as any[], world(duskFx, [27274249, 25704967, 31982592, 12605441]));
  } catch (e) {
    total += 1;
    out.push(`P1RUN ${RUN_ID} ERROR ${String(e)}`);
  }
  out.push(`P1RUN ${RUN_ID} done mismatches=${total}`);
  for (const l of out) console.log(l);
  return out.join('\n');
}
const text = go();
try {
  const f = new File(Paths.document, 'p1run.txt');
  f.create({ overwrite: true });
  f.write(text);
} catch (e) {
  console.log('write failed ' + String(e));
}
export default function App() {
  return (
    <View style={{ flex: 1, justifyContent: 'center', padding: 16 }}>
      <Text style={{ fontSize: 10 }}>{text.slice(0, 1500)}</Text>
    </View>
  );
}
