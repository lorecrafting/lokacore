import { Meta, Markdown } from '@storybook/addon-docs/blocks';
import { specs } from './specs.ts';

export const sources = import.meta.glob(['../book/*.tsx', '../SaveError.tsx'], {
  query: '?raw',
  import: 'default',
  eager: true,
});
export const all = Object.entries(sources).flatMap(
  ([path, src]) => specs(path.slice(3), src).components,
);
